def _polydata(vertices, triangles, triangle_ids=None):
    points = vtkPoints()
    points.SetNumberOfPoints(len(vertices))
    for index, vertex in enumerate(vertices):
        points.SetPoint(index, float(vertex.x), float(vertex.y), float(vertex.z))

    cells = vtkCellArray()
    ids = range(len(triangles)) if triangle_ids is None else triangle_ids
    for triangle_id in ids:
        triangle = triangles[triangle_id]
        cells.InsertNextCell(3)
        cells.InsertCellPoint(int(triangle[0]))
        cells.InsertCellPoint(int(triangle[1]))
        cells.InsertCellPoint(int(triangle[2]))

    data = vtkPolyData()
    data.SetPoints(points)
    data.SetPolys(cells)
    return data


def _actor_for_polydata(polydata, color, opacity=1.0, edges=False, offset=False):
    normals = vtkPolyDataNormals()
    normals.SetInputData(polydata)
    normals.ComputePointNormalsOff()
    normals.ComputeCellNormalsOn()
    mapper = vtkPolyDataMapper()
    mapper.SetInputConnection(normals.GetOutputPort())
    mapper.ScalarVisibilityOff()
    if offset:
        mapper.SetResolveCoincidentTopologyToPolygonOffset()
        mapper.SetRelativeCoincidentTopologyPolygonOffsetParameters(-4.0, -4.0)
    actor = vtkActor()
    actor.SetMapper(mapper)
    actor.GetProperty().SetColor(*color)
    actor.GetProperty().SetOpacity(opacity)
    actor.GetProperty().SetEdgeVisibility(bool(edges))
    actor.GetProperty().SetEdgeColor(0.15, 0.17, 0.20)
    actor.GetProperty().SetLineWidth(1.0)
    return actor, mapper


def _line_polydata(vertices, edges):
    points = vtkPoints()
    points.SetNumberOfPoints(len(vertices))
    for index, vertex in enumerate(vertices):
        points.SetPoint(index, float(vertex.x), float(vertex.y), float(vertex.z))
    lines = vtkCellArray()
    for first, second in edges:
        lines.InsertNextCell(2)
        lines.InsertCellPoint(int(first))
        lines.InsertCellPoint(int(second))
    data = vtkPolyData()
    data.SetPoints(points)
    data.SetLines(lines)
    return data

    def __init__(self, mesh, voxel_size=None, initial_selection=None, parent=None):
        super().__init__(parent, title="OpenVCAD Mesh Selection")
        if not isinstance(mesh, pv.SurfaceMesh):
            raise TypeError("mesh must be a pyvcad.SurfaceMesh")
        if initial_selection is not None:
            if not isinstance(initial_selection, pv.MeshSelection):
                raise TypeError("initial_selection must be a pyvcad.MeshSelection")
            if initial_selection.source_fingerprint != mesh.fingerprint:

        self.method_choice.SetStringSelection("Linked by face angle")
        self.action_choice.SetStringSelection("Replace")

    def _build_scene(self):
        self.renderer = vtkRenderer()
        self.renderer.SetBackground(0.96, 0.97, 0.98)
        self.render_window = self.vtk_widget.GetRenderWindow()
        self.render_window.AddRenderer(self.renderer)
        interactor = self.render_window.GetInteractor()
        interactor.SetInteractorStyle(vtkInteractorStyleTrackballCamera())

        self.base_polydata = _polydata(self._vertices, self._triangles)
        self.base_actor, self.base_mapper = _actor_for_polydata(
            self.base_polydata, (0.70, 0.73, 0.78), opacity=1.0, edges=True
        )
        self.selection_actor, self.selection_mapper = _actor_for_polydata(
            _polydata(self._vertices, self._triangles, []),
            (1.0, 0.35, 0.05),
            opacity=1.0,
            edges=True,
            offset=True,
        )
        self.boundary_mapper = vtkPolyDataMapper()
        self.boundary_mapper.SetInputData(_line_polydata(self._vertices, []))
        self.boundary_actor = vtkActor()
        self.boundary_actor.SetMapper(self.boundary_mapper)
        self.boundary_actor.GetProperty().SetColor(0.05, 0.15, 0.25)
        self.boundary_actor.GetProperty().SetLineWidth(4.0)
        self.renderer.AddActor(self.base_actor)
        self.renderer.AddActor(self.selection_actor)
        self.renderer.AddActor(self.boundary_actor)
        self.renderer.ResetCamera()

        self.picker = vtkCellPicker()
        self.picker.SetTolerance(0.002)
        self.picker.InitializePickList()
        self.picker.AddPickList(self.base_actor)
        self.picker.PickFromListOn()
        self._left_press_tag = interactor.AddObserver(
            "LeftButtonPressEvent", self._on_left_button_press, 10.0
        )
        interactor.AddObserver("MouseMoveEvent", self._on_mouse_move, 10.0)
        interactor.AddObserver("LeftButtonReleaseEvent", self._on_left_button_release, 10.0)

    def _on_left_button_press(self, caller, _event):
        if self.interaction_choice.GetSelection() != 0:
            return
        interactor = self.render_window.GetInteractor()
        x, y = interactor.GetEventPosition()
        if not self.picker.Pick(x, y, 0, self.renderer):
            return
        triangle_id = int(self.picker.GetCellId())
        if triangle_id < 0:
            return
        point = tuple(float(value) for value in self.picker.GetPickPosition())
        method = METHOD_IDS[self.method_choice.GetSelection()]
        if method == "manual":
            self._start_paint(triangle_id)
        else:
            self._begin_selection(triangle_id, point)
        command = caller.GetCommand(self._left_press_tag)
        if command is not None:
            command.SetAbortFlag(1)

    def _start_paint(self, triangle_id):
        self._painting = True
        self._paint_base = set(self._selected_ids)
        self._painted_ids = set()
        self._paint_action = ACTION_IDS[self.action_choice.GetSelection()]
        self._paint_triangle(triangle_id)

    def _paint_triangle(self, triangle_id):
        if triangle_id in self._painted_ids:
            return
        self._painted_ids.add(triangle_id)
        if self._paint_action == "replace":
            self._selected_ids = set(self._painted_ids)
        elif self._paint_action == "add":
            self._selected_ids = self._paint_base | self._painted_ids
        else:
            self._selected_ids = self._paint_base - self._painted_ids
        self._recipe = {"method": "manual"}
        self._preview = None
        self._refresh_selection()

    def _on_mouse_move(self, _caller, _event):
        if not self._painting or self.interaction_choice.GetSelection() != 0:
            return
        interactor = self.render_window.GetInteractor()
        x, y = interactor.GetEventPosition()
        if self.picker.Pick(x, y, 0, self.renderer):
            triangle_id = int(self.picker.GetCellId())
            if triangle_id >= 0:
                self._paint_triangle(triangle_id)

    def _on_left_button_release(self, _caller, _event):
        self._painting = False

    def _begin_selection(self, triangle_id, point):
        method = METHOD_IDS[self.method_choice.GetSelection()]
        action = ACTION_IDS[self.action_choice.GetSelection()]
        self._preview = {
            "base": set(self._selected_ids),
            "triangle_id": triangle_id,
            "point": point,
            "method": method,
            "action": action,
        }
        self._apply_preview()

    def _apply_preview(self):
        if self._preview is None:
            return
        method = self._preview["method"]
        triangle_id = self._preview["triangle_id"]
        if method == "manual":
            candidate = self.mesh.select_triangles([triangle_id])
            recipe = {"method": "manual"}
        elif method == "linked":
            candidate = self.mesh.select_linked(triangle_id)
            recipe = mesh_selection_recipe("linked", point=self._preview["point"])
        elif method == "linked_by_face_angle":
            angle = self.angle_control.GetValue()
            candidate = self.mesh.select_linked_by_face_angle(triangle_id, angle)
            recipe = mesh_selection_recipe(
                "linked_by_face_angle", point=self._preview["point"],
                max_angle_degrees=angle
            )
        else:
            radius = self.radius_control.GetValue()
            seed_vertex = self.mesh.nearest_vertex(pv.Vec3(self._preview["point"]))
            candidate = self.mesh.select_by_geodesic_radius(seed_vertex, radius)
            recipe = mesh_selection_recipe(
                "geodesic_radius", point=self._preview["point"], radius=radius
            )

        candidate_ids = set(candidate.triangle_ids)
        base = self._preview["base"]
        action = self._preview["action"]
        if action == "replace":
            self._selected_ids = candidate_ids
            self._recipe = recipe
        elif action == "add":
            self._selected_ids = base | candidate_ids
            self._recipe = recipe if not base else {"method": "manual"}
        else:
            self._selected_ids = base - candidate_ids
            self._recipe = {"method": "manual"}
        self._refresh_selection()

    def _refresh_selection(self, render=True):
        selected = sorted(self._selected_ids)
        self.selection_mapper.SetInputData(_polydata(self._vertices, self._triangles, selected))
        self.selection_mapper.Update()
        if selected:
            selection = self.mesh.select_triangles(selected)
            area = selection.area
            boundary_edges = selection.boundary_edges()
        else:
            area = 0.0
            boundary_edges = []
        self.boundary_mapper.SetInputData(_line_polydata(self._vertices, boundary_edges))
        self.boundary_mapper.Update()
        self.summary_label.SetLabel(
            "{} of {} triangles\nSelected area: {:.3f} mm²".format(
                len(selected), len(self._triangles), area
            )
        )
        self.validation_label.SetLabel("Conformal validation has not been run.")
        self.accept_button.Enable(bool(selected))
        self.copy_button.Enable(bool(selected))
        self.save_button.Enable(bool(selected))
        self.validate_button.Enable(bool(selected))
        if render:
            self.render_window.Render()

    def _on_method_change(self, _event):
        method = METHOD_IDS[self.method_choice.GetSelection()]
        is_angle = method == "linked_by_face_angle"
        is_radius = method == "geodesic_radius"
        self.angle_control.Enable(is_angle)
        self.angle_label.Enable(is_angle)
        self.radius_control.Enable(is_radius)
        self.radius_label.Enable(is_radius)
        self._preview = None

    def _on_angle_change(self, _event):
        if self._preview is not None and self._preview["method"] == "linked_by_face_angle":
            self._apply_preview()

    def _on_radius_change(self, _event):
        if self._preview is not None and self._preview["method"] == "geodesic_radius":
            self._apply_preview()

    def _on_clear(self, _event):
        self._selected_ids.clear()
        self._recipe = {"method": "manual"}
        self._preview = None
        self._refresh_selection()

    def _on_invert(self, _event):
        self._selected_ids = set(range(len(self._triangles))) - self._selected_ids
        self._recipe = {"method": "manual"}
        self._preview = None
        self._refresh_selection()

    def _on_grow(self, _event):
        if not self._selected_ids:
            return
        selection = self.mesh.select_triangles(sorted(self._selected_ids)).grown()
        self._selected_ids = set(selection.triangle_ids)
        self._recipe = {"method": "manual"}
        self._preview = None
        self._refresh_selection()

    def _on_shrink(self, _event):
        if not self._selected_ids:
            return
        selection = self.mesh.select_triangles(sorted(self._selected_ids)).shrunk()
        self._selected_ids = set(selection.triangle_ids)
        self._recipe = {"method": "manual"}
        self._preview = None
        self._refresh_selection()

    def _on_validate(self, _event):
        try:
            surface = self.mesh.select_triangles(
                sorted(self._selected_ids)
            ).to_triangle_mesh_surface()
            self.validation_label.SetLabel(
                "Conformal patch ready: one connected open disk, {} triangles.".format(
                    surface.triangle_count
                )
            )
        except RuntimeError as error:
            self.validation_label.SetLabel("Not a conformal disk: {}".format(error))
        self.validation_label.Wrap(self.FromDIP(275))
        self.Layout()

    def _on_copy(self, _event):
        text = repr(sorted(self._selected_ids))
        if wx.TheClipboard.Open():
            wx.TheClipboard.SetData(wx.TextDataObject(text))
            wx.TheClipboard.Close()
            self.SetStatusText("Copied {} triangle IDs.".format(len(self._selected_ids)))

    def _on_save(self, _event):
        dialog = wx.FileDialog(
            self,
            message="Save mesh selection",
            wildcard="OpenVCAD mesh selection (*.selection.json)|*.selection.json|JSON (*.json)|*.json",
            style=wx.FD_SAVE | wx.FD_OVERWRITE_PROMPT,
        )
        if dialog.ShowModal() != wx.ID_OK:
            dialog.Destroy()
            return
        path = Path(dialog.GetPath())
        dialog.Destroy()
        if not path.name.endswith(".json"):
            path = path.with_name(path.name + ".selection.json")
        try:
            save_mesh_selection(
                self.mesh.select_triangles(sorted(self._selected_ids)),
                path,
                recipe=self._recipe,
                voxel_size=self.voxel_size,
            )
            self.SetStatusText("Saved selection to {}".format(path))
        except (OSError, RuntimeError, ValueError) as error:
            wx.MessageBox(str(error), "Unable to save selection", wx.OK | wx.ICON_ERROR, self)

    def _on_accept(self, _event):
        if not self._selected_ids:
            return
        self.result = self.mesh.select_triangles(sorted(self._selected_ids))
        self._accepted = True
        self.Close()

    def _on_cancel(self, _event):
        self.result = None
        self.Close()

    def _on_close(self, event):
        if not self._accepted:
            self.result = None
        event.Skip()

def select_mesh_patch(source, voxel_size=None, initial_selection=None):
    """Open the auxiliary picker and return a ``pyvcad.MeshSelection`` or ``None``.

    ``source`` may be an existing ``SurfaceMesh`` or an OpenVCAD ``Node``. Tree inputs require an
    explicit voxel size because the selected triangle IDs are resolution-specific.

    Arguments:
      source (SurfaceMesh | Node): The mesh, or the tree to mesh.
      voxel_size (float, optional): Meshing voxel size in mm. Required for a ``Node``.
      initial_selection (MeshSelection, optional): Triangles to show as selected when the picker opens.

    Returns:
      MeshSelection | None: The accepted selection, or None if the picker was closed without one.

    Raises:
      RuntimeError: If called while a wx event loop is already running.
    """
    if wx.App.Get() is not None:
        raise RuntimeError("select_mesh_patch must be launched outside an existing wx event loop")
    mesh = resolve_selection_mesh(source, voxel_size=voxel_size)
    with wx.LogNull():
        app = wx.App(False)
        app.frame = MeshSelectionFrame(
            mesh, voxel_size=voxel_size, initial_selection=initial_selection
        )
    app.MainLoop()
    return app.frame.result
